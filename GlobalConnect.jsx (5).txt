// ─── MAIN APP ──────────────────────────────────────────────────────────────────
export default function GlobalConnect() {
  const [activeTab, setActiveTab] = useState("feed");
  const [modal, setModal] = useState(null);
  const [viewProfile, setViewProfile] = useState(null);
  const notifCount = NOTIFICATIONS.filter(n => !n.read).length;
  const msgCount = MOCK_MESSAGES.reduce((a, m) => a + m.unread, 0);

  const renderMain = () => {
    switch (activeTab) {
      case "feed": return <FeedTab />;
      case "explore": return <ExploreTab setActiveTab={setActiveTab} setViewProfile={setViewProfile} />;
      case "reels": return <ReelsTab />;
      case "messages": return <MessagesTab />;
      case "groups": return <GroupsTab />;
      case "marketplace": return <MarketplaceTab setModal={setModal} />;
      case "profile": return <ProfileTab user={CURRENT_USER} currentUser={CURRENT_USER} />;
      case "viewprofile": return <ProfileTab user={viewProfile || CURRENT_USER} currentUser={CURRENT_USER} />;
      default: return <FeedTab />;
    }
  };

  return (
    <>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Sora:wght@400;500;600;700;800&display=swap');
        * { margin: 0; padding: 0; box-sizing: border-box; }
        body { background: #0A1628; }
        ::-webkit-scrollbar { width: 4px; height: 4px; }
        ::-webkit-scrollbar-track { background: transparent; }
        ::-webkit-scrollbar-thumb { background: rgba(0,212,168,0.3); border-radius: 4px; }
        input::placeholder, textarea::placeholder { color: rgba(255,255,255,0.3); }
      `}</style>
      <div style={{ background: "#0A1628", minHeight: "100vh", fontFamily: "'Sora',sans-serif" }}>
        <Navbar activeTab={activeTab} setActiveTab={setActiveTab} notifCount={notifCount} msgCount={msgCount} setModal={setModal} />

        {/* Bottom nav (mobile) */}
        <div style={{ position: "fixed", bottom: 0, left: 0, right: 0, background: "rgba(8,20,40,0.97)", backdropFilter: "blur(20px)", borderTop: "1px solid rgba(0,212,168,0.12)", display: "flex", zIndex: 999, padding: "6px 0" }}>
          {[["⊞","feed"],["🔍","explore"],["▶","reels"],["🛒","marketplace"],["👤","profile"]].map(([icon,tab]) => (
            <button key={tab} onClick={() => setActiveTab(tab)} style={{ flex: 1, background: "transparent", border: "none", color: activeTab === tab ? "#00D4A8" : "rgba(255,255,255,0.4)", cursor: "pointer", display: "flex", flexDirection: "column", alignItems: "center", gap: 2, padding: "4px 0" }}>
              <span style={{ fontSize: 20 }}>{icon}</span>
              <span style={{ fontSize: 9, fontFamily: "'Sora',sans-serif", textTransform: "capitalize" }}>{tab}</span>
            </button>
          ))}
        </div>

        {/* Main layout */}
        <div style={{ maxWidth: 1100, margin: "0 auto", padding: "68px 12px 70px", display: "flex", gap: 16 }}>
          {/* Left sidebar — desktop only */}
          <div style={{ display: window.innerWidth >= 900 ? "block" : "none" }}>
            <LeftSidebar setActiveTab={setActiveTab} activeTab={activeTab} setModal={setModal} />
          </div>

          {/* Main content */}
          <div style={{ flex: 1, minWidth: 0 }}>
            {renderMain()}
          </div>

          {/* Right sidebar — desktop only */}
          <div style={{ display: window.innerWidth >= 1024 ? "block" : "none" }}>
            <RightSidebar setActiveTab={setActiveTab} setViewProfile={setViewProfile} />
          </div>
        </div>

        {/* Modals */}
        {modal === "data" && <DataModal onClose={() => setModal(null)} />}
        {modal === "advertise" && <AdvertiseModal onClose={() => setModal(null)} />}
        {modal === "notifications" && <NotificationsModal onClose={() => setModal(null)} />}
      </div>
    </>
  );
}
